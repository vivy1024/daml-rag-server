## 🔧 一键部署脚本

### 创建快速部署脚本

```python
#!/usr/bin/env python3
"""
一键部署知识图谱到Neo4j
适用于云服务器首次部署

使用：
    python deploy_kg_to_cloud.py [方案]
    
方案选项：
    minimal     - 最小部署（5-15分钟）
    recommended - 推荐部署（10-25分钟，默认）
    full        - 完整部署（20-40分钟）
"""

import sys
import subprocess
import time
from pathlib import Path
from datetime import datetime

# 脚本目录
SCRIPTS_DIR = Path(__file__).parent / "scripts"

# 部署方案定义
DEPLOY_PLANS = {
    "minimal": [
        ("create_kg_schema.py", "创建Schema"),
        ("migrate_neo4j_only.py", "导入Exercise数据"),
        ("import_nutrition.py", "导入营养数据"),
    ],
    "recommended": [
        ("create_kg_schema.py", "创建Schema"),
        ("migrate_neo4j_only.py", "导入Exercise数据"),
        ("import_nutrition.py", "导入营养数据"),
        ("import_training_programs.py", "导入训练计划"),
        ("create_crossdomain_relations.py", "创建跨域关系"),
        ("supplement_exercise_relations.py", "补充Exercise关系"),
    ],
    "full": [
        ("create_kg_schema.py", "创建Schema"),
        ("migrate_neo4j_only.py", "导入Exercise数据"),
        ("import_nutrition.py", "导入营养数据"),
        ("import_training_programs.py", "导入训练计划"),
        ("create_crossdomain_relations.py", "创建跨域关系"),
        ("supplement_exercise_relations.py", "补充Exercise关系"),
        ("import_diseases.py", "导入疾病数据"),
        ("import_rpe_zones.py", "导入RPE区间"),
        ("import_training_principles.py", "导入训练原则"),
        ("create_rehab_schema.py", "创建康复Schema"),
    ],
}

def run_script(script_name: str, description: str) -> bool:
    """运行单个导入脚本"""
    script_path = SCRIPTS_DIR / script_name
    
    if not script_path.exists():
        print(f"  ⚠️  脚本不存在: {script_name}")
        return False
    
    print(f"\n{'='*60}")
    print(f"  📝 {description}")
    print(f"  🔧 执行: {script_name}")
    print(f"{'='*60}")
    
    start_time = time.time()
    
    try:
        result = subprocess.run(
            [sys.executable, str(script_path)],
            capture_output=True,
            text=True,
            encoding='utf-8'
        )
        
        elapsed = time.time() - start_time
        
        if result.returncode == 0:
            print(f"  ✅ 成功！耗时: {elapsed:.1f}秒")
            if result.stdout:
                print(f"\n{result.stdout}")
            return True
        else:
            print(f"  ❌ 失败！错误码: {result.returncode}")
            if result.stderr:
                print(f"\n错误信息:\n{result.stderr}")
            return False
            
    except Exception as e:
        print(f"  ❌ 执行异常: {e}")
        return False

def deploy(plan_name: str = "recommended"):
    """执行部署"""
    
    if plan_name not in DEPLOY_PLANS:
        print(f"❌ 未知方案: {plan_name}")
        print(f"可用方案: {', '.join(DEPLOY_PLANS.keys())}")
        return False
    
    plan = DEPLOY_PLANS[plan_name]
    
    print("=" * 80)
    print(f"  🚀 BUILD_BODY 知识图谱云端部署")
    print(f"  📦 部署方案: {plan_name.upper()}")
    print(f"  📊 脚本数量: {len(plan)}个")
    print(f"  ⏰ 开始时间: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    print("=" * 80)
    
    total_start = time.time()
    success_count = 0
    failed_scripts = []
    
    for i, (script, desc) in enumerate(plan, 1):
        print(f"\n\n[{i}/{len(plan)}] 执行中...")
        
        if run_script(script, desc):
            success_count += 1
        else:
            failed_scripts.append((script, desc))
            
            # 询问是否继续
            if i < len(plan):
                response = input("\n⚠️  脚本执行失败，是否继续？(y/n): ")
                if response.lower() != 'y':
                    break
    
    total_elapsed = time.time() - total_start
    
    # 总结报告
    print("\n\n" + "=" * 80)
    print(f"  🎉 部署完成报告")
    print("=" * 80)
    print(f"  ✅ 成功: {success_count}/{len(plan)}个脚本")
    print(f"  ⏱️  总耗时: {total_elapsed/60:.1f}分钟")
    print(f"  📅 完成时间: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
    
    if failed_scripts:
        print(f"\n  ❌ 失败脚本:")
        for script, desc in failed_scripts:
            print(f"     - {script} ({desc})")
    else:
        print(f"\n  🎊 所有脚本执行成功！")
    
    print("=" * 80)
    
    return len(failed_scripts) == 0

if __name__ == "__main__":
    plan = sys.argv[1] if len(sys.argv) > 1 else "recommended"
    success = deploy(plan)
    sys.exit(0 if success else 1)
```

## ⚠️ 重要注意事项

### 1. Neo4j连接配置

所有导入脚本默认使用以下配置：
```python
uri = 'bolt://localhost:7687'
user = 'neo4j'
password = 'password'  # ⚠️ 需要改为 'fitness123'
```

**修复方法**：
- 方案A：修改各脚本的密码为 `fitness123`
- 方案B：修改Neo4j密码为 `password`
- 方案C：使用环境变量 `NEO4J_PASSWORD=fitness123`

---

### 2. Docker环境连接

如果Neo4j运行在Docker中，需要修改URI：
```python
# 从容器内访问
uri = 'bolt://neo4j:7687'

# 从宿主机访问
uri = 'bolt://localhost:7687'
```

---

### 3. 数据文件路径

确保以下文件存在：
```bash
# 必需文件
mcp-servers/fitness_daml_rag/data/knowledge_graph_mvp.db
mcp-servers/fitness_daml_rag/data/training_programs.json

# 可选文件（用于完整部署）
mcp-servers/fitness_daml_rag/data/chinese_nutrition_db.json
mcp-servers/archive/nutrition-guide-stdio/data/*.json
```

---

### 4. 导入顺序

**严格按照以下顺序**：
1. ✅ 先创建Schema (`create_kg_schema.py`)
2. ✅ 再导入节点数据（Exercise, Food等）
3. ✅ 最后创建关系（crossdomain, supplement等）

**错误的顺序会导致**：
- ❌ 节点创建失败
- ❌ 关系创建失败
- ❌ 外键约束错误
