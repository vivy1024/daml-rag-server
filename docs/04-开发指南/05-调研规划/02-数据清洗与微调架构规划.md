#### 规则 2：去除重复查询

```python
def deduplicate_by_similarity(conversations, threshold=0.95):
    """
    基于语义相似度去重
    
    Args:
        conversations: 对话列表
        threshold: 相似度阈值（>= 则判定为重复）
    
    Returns:
        去重后的对话列表
    """
    from sentence_transformers import SentenceTransformer
    import numpy as np
    
    model = SentenceTransformer('BAAI/bge-base-zh-v1.5')
    
    # 编码所有查询
    queries = [conv['query'] for conv in conversations]
    embeddings = model.encode(queries, normalize_embeddings=True)
    
    # 计算相似度矩阵
    similarity_matrix = np.dot(embeddings, embeddings.T)
    
    # 去重（保留评分最高的）
    deduplicated = []
    used_indices = set()
    
    for i in range(len(conversations)):
        if i in used_indices:
            continue
        
        # 找到所有与当前查询相似的索引
        similar_indices = np.where(similarity_matrix[i] >= threshold)[0]
        
        # 选择评分最高的
        best_idx = max(similar_indices, key=lambda idx: conversations[idx]['reward'])
        deduplicated.append(conversations[best_idx])
        
        # 标记已使用
        used_indices.update(similar_indices)
    
    return deduplicated
```

**去重策略**：
- 相似度 **≥ 0.95** → 判定为重复
- 保留评分最高的对话
- 记录去重统计信息

#### 规则 3：标注工具链

```python
def extract_tool_chain(tools_used_str):
    """
    从 tools_used 字段提取工具链
    
    Args:
        tools_used_str: "tool1,tool2,tool3" 或 JSON字符串
    
    Returns:
        List[str]: 工具链列表
    """
    import json
    
    try:
        # 尝试解析JSON
        if tools_used_str.startswith('['):
            return json.loads(tools_used_str)
        else:
            # 逗号分隔
            return [tool.strip() for tool in tools_used_str.split(',')]
    except:
        return []

def classify_domain(query):
    """
    分类查询领域
    
    Args:
        query: 用户查询
    
    Returns:
        str: 领域标签
    """
    keywords = {
        "muscle_gain": ["增肌", "增重", "增加肌肉", "长肌肉", "力量", "muscle gain"],
        "fat_loss": ["减脂", "减肥", "减重", "瘦身", "燃脂", "fat loss"],
        "rehabilitation": ["康复", "恢复", "损伤", "受伤", "疼痛", "rehabilitation"],
        "nutrition": ["营养", "饮食", "食谱", "吃什么", "补充剂", "nutrition"],
        "general": []  # 通用/其他
    }
    
    for domain, kws in keywords.items():
        for kw in kws:
            if kw in query.lower():
                return domain
    
    return "general"
```

#### 规则 4：数据验证

```python
def validate_conversation(conv):
    """
    验证对话数据完整性
    
    Returns:
        Tuple[bool, str]: (是否有效, 错误原因)
    """
    # 必填字段检查
    required_fields = ['query', 'response', 'tools_used', 'reward']
    for field in required_fields:
        if not conv.get(field):
            return False, f"缺少必填字段: {field}"
    
    # 工具链检查
    if len(conv['tools_used']) == 0:
        return False, "工具链为空"
    
    # 响应长度检查
    if len(conv['response']) < 50:
        return False, "响应内容过短"
    
    # 评分范围检查
    if not (1 <= conv['reward'] <= 5):
        return False, "评分超出范围 [1, 5]"
    
    return True, ""
```

**LoRA (Low-Rank Adaptation)** 参数：

```python
from peft import LoraConfig, get_peft_model

lora_config = LoraConfig(
    r=16,                # Rank（秩）：控制适配器的参数量
    lora_alpha=32,       # Alpha：缩放因子（通常为 r 的 2 倍）
    lora_dropout=0.05,   # Dropout：防止过拟合
    target_modules=[     # 目标模块：对哪些层应用LoRA
        "q_proj",
        "v_proj",
        "k_proj",
        "o_proj",
        "gate_proj",
        "up_proj",
        "down_proj"
    ],
    bias="none",         # 不训练bias
    task_type="CAUSAL_LM"  # 任务类型：因果语言模型
)
```

**为什么选择 LoRA？**
- ✅ 参数量小（<1% 原模型参数）
- ✅ 训练快速（GPU时间减少80%+）
- ✅ 部署灵活（可切换适配器）
- ✅ 避免灾难性遗忘

### 5.2 评估脚本

```python
def evaluate_model(model, test_dataset):
    """
    评估微调后的模型
    
    Returns:
        Dict: 评估结果
    """
    results = {
        "user_satisfaction": [],
        "tool_chain_accuracy": [],
        "response_time": [],
        "few_shot_hits": 0
    }
    
    for example in test_dataset:
        # 1. 生成预测
        start_time = time.time()
        prediction = model.predict(example['input'])
        response_time = time.time() - start_time
        
        # 2. 工具链准确率
        predicted_tools = extract_tool_chain(prediction)
        ground_truth_tools = example['metadata']['tool_chain']
        tool_chain_accuracy = jaccard_similarity(predicted_tools, ground_truth_tools)
        
        # 3. 响应质量（人工评分 or GPT-4评分）
        quality_score = rate_response_quality(prediction, example['output'])
        
        # 记录结果
        results["tool_chain_accuracy"].append(tool_chain_accuracy)
        results["response_time"].append(response_time)
        results["user_satisfaction"].append(quality_score)
    
    # 汇总统计
    return {
        "avg_user_satisfaction": np.mean(results["user_satisfaction"]),
        "tool_chain_accuracy": np.mean(results["tool_chain_accuracy"]),
        "p95_response_time": np.percentile(results["response_time"], 95),
        "sample_size": len(test_dataset)
    }
```

## 6. 风险评估与缓解

### 6.1 风险清单

| 风险类型 | 具体描述 | 概率 | 影响 | 缓解措施 |
|---------|---------|------|-----|---------|
| **数据质量** | 清洗后数据仍有噪声 | 中 | 高 | 人工抽查100条 + 严格验证规则 |
| **过拟合** | 模型在训练集上表现好，测试集差 | 中 | 中 | Early stopping + 验证集监控 + Dropout |
| **性能降低** | 微调后响应时间变长 | 低 | 高 | 模型量化 + 推理优化 + A/B测试 |
| **场景覆盖不足** | 某些场景数据过少 | 中 | 中 | 主动生成补充数据 + 迁移学习 |
| **灾难性遗忘** | 微调后通用能力下降 | 低 | 中 | LoRA方法 + 保留基础任务数据 |
| **GPU资源不足** | 训练时GPU不可用/成本超预算 | 低 | 中 | 租用云GPU（AWS/Azure） + 训练队列 |
| **模型泄漏** | 微调模型被盗用 | 低 | 高 | 访问控制 + 模型加密 + 审计日志 |

### 6.2 缓解措施详细说明

# 人工抽查流程
def manual_review_sample(dataset, sample_size=100):
    """
    人工抽查样本质量
    """
    sample = random.sample(dataset, sample_size)
    
    for idx, example in enumerate(sample, 1):
        print(f"--- 样本 {idx}/{sample_size} ---")
        print(f"查询: {example['query']}")
        print(f"响应: {example['response']}")
        print(f"工具链: {example['tool_chain']}")
        print(f"评分: {example['reward']}")
        
        # 人工评估
        quality = input("质量评估（1-5）: ")
        is_correct_tool_chain = input("工具链正确？(y/n): ")
        
        # 记录评估结果
        example['manual_quality'] = int(quality)
        example['manual_tool_chain_correct'] = (is_correct_tool_chain.lower() == 'y')
    
    # 统计
    avg_quality = np.mean([e['manual_quality'] for e in sample])
    tool_chain_accuracy = sum(e['manual_tool_chain_correct'] for e in sample) / len(sample)
    
    print(f"\n人工评估结果：")
    print(f"平均质量: {avg_quality:.2f}")
    print(f"工具链准确率: {tool_chain_accuracy:.1%}")
    
    return avg_quality >= 4.0 and tool_chain_accuracy >= 0.80
```

    ├─ 是 → 继续
    └─ 否 → 优化清洗规则，延迟1周

决策点2（+3个月）：是否达到微调触发条件？
    ├─ 是 → 启动微调训练
    └─ 否 → 继续积累数据，延迟1个月

决策点3（+4个月）：A/B测试结果如何？
    ├─ 优于基线 → 进入灰度发布
    └─ 劣于基线 → 回滚，重新训练

决策点4（+4.5个月）：灰度发布是否稳定？
    ├─ 是 → 全量替换
    └─ 否 → 降级到基线模型，诊断问题
```
