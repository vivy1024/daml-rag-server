**版本**: v1.0.0  
**创建日期**: 2025-12-22

## 概述

本指南说明如何使用DAML-RAG系统的流式输出（Server-Sent Events, SSE）功能。流式输出允许服务器向客户端实时推送数据，实现渐进式内容展示和实时进度反馈，大幅提升用户体验。

### 核心优势

- ✅ **实时进度反馈**：每个步骤完成后立即推送进度
- ✅ **渐进式内容展示**：LLM生成的内容逐字推送
- ✅ **错误实时通知**：错误发生时立即通知客户端
- ✅ **性能监控集成**：实时推送性能指标
- ✅ **优化TTFB**：首字节时间<200ms，提升用户体验

## 前端集成示例

### 方法1：使用EventSource（推荐）

**适用场景**：现代浏览器，简单易用

```javascript
// 创建EventSource连接
const eventSource = new EventSource('/api/chat/stream', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json'
  },
  body: JSON.stringify({
    query: '我想制定一个完整的增肌训练计划',
    user_id: 'user_123',
    session_id: 'session_456'
  })
});

// 监听进度事件
eventSource.addEventListener('progress', (event) => {
  const data = JSON.parse(event.data);
  console.log(`步骤${data.step}: ${data.step_name} - ${data.status}`);
  
  // 更新进度条
  updateProgressBar(data.step, 11);
  
  // 显示步骤信息
  displayStepInfo(data);
});

// 监听内容事件（LLM生成的内容）
eventSource.addEventListener('content', (event) => {
  const data = JSON.parse(event.data);
  
  // 逐字追加内容
  appendContent(data.content);
  
  // 如果是最终内容
  if (data.is_final) {
    finalizeContent();
  }
});

// 监听错误事件
eventSource.addEventListener('error', (event) => {
  const data = JSON.parse(event.data);
  console.error(`错误: ${data.error}`);
  
  // 显示错误信息
  showError(data.error, data.step);
  
  // 关闭连接
  eventSource.close();
});

// 监听性能指标事件
eventSource.addEventListener('metrics', (event) => {
  const data = JSON.parse(event.data);
  console.log('性能指标:', data);
  
  // 显示性能信息
  displayMetrics(data);
});

// 监听完成事件
eventSource.addEventListener('done', (event) => {
  const data = JSON.parse(event.data);
  console.log('完成！总耗时:', data.total_duration_ms);
  
  // 显示完成信息
  displayCompletion(data);
  
  // 关闭连接
  eventSource.close();
});

// 处理连接错误
eventSource.onerror = (error) => {
  console.error('连接错误:', error);
  eventSource.close();
  
  // 可选：自动重试
  setTimeout(() => {
    retryConnection();
  }, 3000);
};
```

### 方法2：使用Fetch API

**适用场景**：需要更多控制，支持自定义请求头

```javascript
async function streamChat(query, userId, sessionId) {
  const response = await fetch('/api/chat/stream', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      query: query,
      user_id: userId,
      session_id: sessionId
    })
  });

  const reader = response.body.getReader();
  const decoder = new TextDecoder();
  let buffer = '';

  while (true) {
    const { done, value } = await reader.read();
    
    if (done) break;
    
    buffer += decoder.decode(value, { stream: true });
    
    // 处理完整的事件
    const events = buffer.split('\n\n');
    buffer = events.pop(); // 保留不完整的事件
    
    for (const event of events) {
      if (!event.trim()) continue;
      
      const lines = event.split('\n');
      let eventType = 'message';
      let eventData = '';
      
      for (const line of lines) {
        if (line.startsWith('event:')) {
          eventType = line.substring(6).trim();
        } else if (line.startsWith('data:')) {
          eventData = line.substring(5).trim();
        }
      }
      
      if (eventData) {
        handleEvent(eventType, JSON.parse(eventData));
      }
    }
  }
}

function handleEvent(type, data) {
  switch (type) {
    case 'progress':
      updateProgressBar(data.step, 11);
      displayStepInfo(data);
      break;
    case 'content':
      appendContent(data.content);
      break;
    case 'error':
      showError(data.error, data.step);
      break;
    case 'metrics':
      displayMetrics(data);
      break;
    case 'done':
      displayCompletion(data);
      break;
  }
}
```

### 方法3：使用Vue.js集成

**适用场景**：Vue.js应用

```vue
<template>
  <div class="chat-container">
    <!-- 进度条 -->
    <div class="progress-bar">
      <div class="progress" :style="{ width: progressPercent + '%' }"></div>
      <span>{{ currentStep }}/11 - {{ currentStepName }}</span>
    </div>
    
    <!-- 内容显示 -->
    <div class="content" v-html="content"></div>
    
    <!-- 错误显示 -->
    <div v-if="error" class="error">{{ error }}</div>
    
    <!-- 性能指标 -->
    <div v-if="showMetrics" class="metrics">
      <span>总耗时: {{ totalDuration }}ms</span>
      <span>缓存命中率: {{ cacheHitRate }}%</span>
    </div>
  </div>
</template>

<script>
export default {
  data() {
    return {
      eventSource: null,
      currentStep: 0,
      currentStepName: '',
      content: '',
      error: null,
      totalDuration: 0,
      cacheHitRate: 0,
      showMetrics: false
    };
  },
  computed: {
    progressPercent() {
      return (this.currentStep / 11) * 100;
    }
  },
  methods: {
    startStream(query, userId, sessionId) {
      this.eventSource = new EventSource('/api/chat/stream', {
        method: 'POST',
        body: JSON.stringify({ query, user_id: userId, session_id: sessionId })
      });
      
      this.eventSource.addEventListener('progress', (event) => {
        const data = JSON.parse(event.data);
        this.currentStep = data.step;
        this.currentStepName = data.step_name;
      });
      
      this.eventSource.addEventListener('content', (event) => {
        const data = JSON.parse(event.data);
        this.content += data.content;
      });
      
      this.eventSource.addEventListener('error', (event) => {
        const data = JSON.parse(event.data);
        this.error = data.error;
        this.eventSource.close();
      });
      
      this.eventSource.addEventListener('metrics', (event) => {
        const data = JSON.parse(event.data);
        this.totalDuration = data.total_duration_ms;
        this.cacheHitRate = (data.cache_hit_rate * 100).toFixed(1);
        this.showMetrics = true;
      });
      
      this.eventSource.addEventListener('done', (event) => {
        const data = JSON.parse(event.data);
        this.eventSource.close();
      });
    },
    stopStream() {
      if (this.eventSource) {
        this.eventSource.close();
      }
    }
  },
  beforeUnmount() {
    this.stopStream();
  }
};
</script>
```

### 方法4：使用React集成

**适用场景**：React应用

```jsx
import React, { useState, useEffect, useRef } from 'react';

function ChatStream({ query, userId, sessionId }) {
  const [currentStep, setCurrentStep] = useState(0);
  const [currentStepName, setCurrentStepName] = useState('');
  const [content, setContent] = useState('');
  const [error, setError] = useState(null);
  const [metrics, setMetrics] = useState(null);
  const eventSourceRef = useRef(null);

  useEffect(() => {
    // 创建EventSource连接
    eventSourceRef.current = new EventSource('/api/chat/stream', {
      method: 'POST',
      body: JSON.stringify({ query, user_id: userId, session_id: sessionId })
    });

    // 监听进度事件
    eventSourceRef.current.addEventListener('progress', (event) => {
      const data = JSON.parse(event.data);
      setCurrentStep(data.step);
      setCurrentStepName(data.step_name);
    });

    // 监听内容事件
    eventSourceRef.current.addEventListener('content', (event) => {
      const data = JSON.parse(event.data);
      setContent(prev => prev + data.content);
    });

    // 监听错误事件
    eventSourceRef.current.addEventListener('error', (event) => {
      const data = JSON.parse(event.data);
      setError(data.error);
      eventSourceRef.current.close();
    });

    // 监听性能指标事件
    eventSourceRef.current.addEventListener('metrics', (event) => {
      const data = JSON.parse(event.data);
      setMetrics(data);
    });

    // 监听完成事件
    eventSourceRef.current.addEventListener('done', (event) => {
      eventSourceRef.current.close();
    });

    // 清理函数
    return () => {
      if (eventSourceRef.current) {
        eventSourceRef.current.close();
      }
    };
  }, [query, userId, sessionId]);

  return (
    <div className="chat-container">
      {/* 进度条 */}
      <div className="progress-bar">
        <div className="progress" style={{ width: `${(currentStep / 11) * 100}%` }}></div>
        <span>{currentStep}/11 - {currentStepName}</span>
      </div>

      {/* 内容显示 */}
      <div className="content" dangerouslySetInnerHTML={{ __html: content }}></div>

      {/* 错误显示 */}
      {error && <div className="error">{error}</div>}

      {/* 性能指标 */}
      {metrics && (
        <div className="metrics">
          <span>总耗时: {metrics.total_duration_ms}ms</span>
          <span>缓存命中率: {(metrics.cache_hit_rate * 100).toFixed(1)}%</span>
        </div>
      )}
    </div>
  );
}

export default ChatStream;
```

## 事件类型详解

### 1. progress - 进度事件

**触发时机**：每个步骤开始和完成时

**数据格式**：
```json
{
  "step": 1,
  "step_name": "预加载用户档案",
  "status": "completed",
  "duration_ms": 120.5,
  "message": "用户档案已加载"
}
```

**字段说明**：
- `step`: 步骤编号（1-11）
- `step_name`: 步骤名称
- `status`: 状态（running/completed/failed）
- `duration_ms`: 执行时长（毫秒）
- `message`: 步骤消息

**使用示例**：
```javascript
eventSource.addEventListener('progress', (event) => {
  const data = JSON.parse(event.data);
  
  if (data.status === 'running') {
    console.log(`开始执行: ${data.step_name}`);
  } else if (data.status === 'completed') {
    console.log(`完成: ${data.step_name} (${data.duration_ms}ms)`);
  } else if (data.status === 'failed') {
    console.error(`失败: ${data.step_name}`);
  }
});
```

### 2. content - 内容事件

**触发时机**：LLM生成内容时（逐字推送）

**数据格式**：
```json
{
  "content": "根据您的",
  "is_final": false
}
```

**字段说明**：
- `content`: 内容片段
- `is_final`: 是否是最终内容

**使用示例**：
```javascript
let fullContent = '';

eventSource.addEventListener('content', (event) => {
  const data = JSON.parse(event.data);
  
  // 追加内容
  fullContent += data.content;
  document.getElementById('content').textContent = fullContent;
  
  // 如果是最终内容
  if (data.is_final) {
    console.log('内容生成完成');
    finalizeContent();
  }
});
```

### 3. error - 错误事件

**触发时机**：发生错误时

**数据格式**：
```json
{
  "error": "工具调用失败",
  "step": 7,
  "details": "intelligent_exercise_selector timeout"
}
```

**字段说明**：
- `error`: 错误消息
- `step`: 发生错误的步骤
- `details`: 错误详情

**使用示例**：
```javascript
eventSource.addEventListener('error', (event) => {
  const data = JSON.parse(event.data);
  
  // 显示错误信息
  showErrorNotification(`步骤${data.step}失败: ${data.error}`);
  
  // 记录错误详情
  console.error('错误详情:', data.details);
  
  // 关闭连接
  eventSource.close();
  
  // 可选：自动重试
  if (shouldRetry(data.error)) {
    setTimeout(() => retryRequest(), 3000);
  }
});
```

### 4. metrics - 性能指标事件

**触发时机**：工作流程完成时

**数据格式**：
```json
{
  "total_duration_ms": 2850.5,
  "steps_completed": 11,
  "cache_hit_rate": 0.85,
  "success_rate": 1.0,
  "ttfb_ms": 180.5
}
```

**字段说明**：
- `total_duration_ms`: 总执行时长（毫秒）
- `steps_completed`: 完成的步骤数
- `cache_hit_rate`: 缓存命中率（0-1）
- `success_rate`: 成功率（0-1）
- `ttfb_ms`: 首字节时间（毫秒）

**使用示例**：
```javascript
eventSource.addEventListener('metrics', (event) => {
  const data = JSON.parse(event.data);
  
  // 显示性能指标
  displayMetrics({
    duration: `${(data.total_duration_ms / 1000).toFixed(2)}秒`,
    cacheHitRate: `${(data.cache_hit_rate * 100).toFixed(1)}%`,
    ttfb: `${data.ttfb_ms.toFixed(0)}ms`
  });
  
  // 记录到分析系统
  analytics.track('workflow_completed', data);
});
```

### 5. done - 完成事件

**触发时机**：流式输出结束时

**数据格式**：
```json
{
  "success": true,
  "total_duration_ms": 2850.5,
  "final_message": "工作流程执行完成"
}
```

**字段说明**：
- `success`: 是否成功
- `total_duration_ms`: 总执行时长（毫秒）
- `final_message`: 最终消息

**使用示例**：
```javascript
eventSource.addEventListener('done', (event) => {
  const data = JSON.parse(event.data);
  
  if (data.success) {
    console.log('✅ 完成！总耗时:', data.total_duration_ms);
    showSuccessNotification(data.final_message);
  } else {
    console.error('❌ 失败');
    showErrorNotification('工作流程执行失败');
  }
  
  // 关闭连接
  eventSource.close();
  
  // 清理资源
  cleanup();
});
```

## 错误处理

### 常见错误类型

1. **连接错误**：网络中断、超时
2. **执行错误**：工具调用失败、LLM调用失败
3. **数据错误**：数据格式错误、解析失败

### 错误处理策略

```javascript
// 错误处理函数
function handleStreamError(error, eventSource) {
  console.error('流式输出错误:', error);
  
  // 关闭连接
  if (eventSource) {
    eventSource.close();
  }
  
  // 显示错误信息
  showErrorNotification('连接中断，请重试');
  
  // 自动重试（最多3次）
  if (retryCount < 3) {
    retryCount++;
    setTimeout(() => {
      console.log(`重试第${retryCount}次...`);
      startStream();
    }, 3000);
  } else {
    console.error('重试次数已达上限');
    showErrorNotification('连接失败，请稍后再试');
  }
}

// 监听连接错误
eventSource.onerror = (error) => {
  handleStreamError(error, eventSource);
};

// 监听error事件
eventSource.addEventListener('error', (event) => {
  const data = JSON.parse(event.data);
  handleStreamError(data.error, eventSource);
});
```

### 超时处理

```javascript
// 设置超时
const timeout = setTimeout(() => {
  console.error('请求超时');
  eventSource.close();
  showErrorNotification('请求超时，请重试');
}, 60000); // 60秒超时

// 完成时清除超时
eventSource.addEventListener('done', () => {
  clearTimeout(timeout);
});
```

## 性能优化

### 1. 连接复用

```javascript
// 复用EventSource连接
let globalEventSource = null;

function getEventSource() {
  if (!globalEventSource || globalEventSource.readyState === EventSource.CLOSED) {
    globalEventSource = new EventSource('/api/chat/stream');
  }
  return globalEventSource;
}
```

### 2. 内容缓冲

```javascript
// 缓冲内容，减少DOM更新
let contentBuffer = '';
let updateTimer = null;

eventSource.addEventListener('content', (event) => {
  const data = JSON.parse(event.data);
  contentBuffer += data.content;
  
  // 每100ms更新一次DOM
  if (!updateTimer) {
    updateTimer = setTimeout(() => {
      document.getElementById('content').textContent = contentBuffer;
      updateTimer = null;
    }, 100);
  }
});
```

### 3. 虚拟滚动

```javascript
// 对于长内容，使用虚拟滚动
import VirtualScroll from 'virtual-scroll-library';

const virtualScroll = new VirtualScroll({
  container: document.getElementById('content'),
  itemHeight: 20,
  bufferSize: 10
});

eventSource.addEventListener('content', (event) => {
  const data = JSON.parse(event.data);
  virtualScroll.append(data.content);
});
```
