### 交互记录

```python
async def _save_interaction(
    self,
    user_id: str,
    session_id: str,
    user_message: str,
    ai_response: str,
    metadata: Dict[str, Any]
):
    """
    保存交互记录
    
    记录内容：
    - 用户查询和AI响应
    - 使用的模型（DeepSeek/Anthropic Claude）
    - 调用的工具列表
    - 质量评分
    - 用户评分（等待用户反馈）
    """
    # 1. 保存到MySQL（结构化）
    await self.backend_client.save_chat_session({
        "user_id": user_id,
        "session_id": session_id,
        "user_message": user_message,
        "ai_response": ai_response,
        "model_used": metadata.get("model_used"),
        "tools_called": metadata.get("tools_called", []),
        "metadata": {
            "few_shot_count": metadata.get("few_shot_count"),
            "similarity_score": metadata.get("similarity"),
            "model_selection_factor": metadata.get("selection_factor")
        }
    })
    
    # 2. 向量化并保存到Qdrant（Few-Shot检索）
    query_vector = self.embedding_model.encode(user_message)
    await self.qdrant_helper.add_conversation({
        "user_query": user_message,
        "ai_response": ai_response,
        "vector": query_vector.tolist(),
        "metadata": {
            "user_id": user_id,
            "user_rating": None,  # 等待用户评分
            "quality_score": self._evaluate_quality(ai_response),
            "model_used": metadata.get("model_used")
        }
    })
```
