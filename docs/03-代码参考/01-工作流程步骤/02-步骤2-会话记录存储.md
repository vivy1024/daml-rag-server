## 概述

步骤2负责将用户与AI的对话记录存储到两个数据库中：
1. **MySQL**：结构化存储，用于查询和分析
2. **Qdrant**：向量化存储，用于Few-Shot检索（步骤6）

这种**双写策略**确保了数据的结构化查询能力和语义检索能力。

### 在11步工作流程中的位置

```
步骤1: 预加载用户档案
  ↓

```

### 双写策略实现

```python
# src/applications/fitness/fitness_orchestrator.py

class FitnessOrchestrator:
    async def save_interaction(
        self,
        user_id: str,
        session_id: str,
        user_message: str,
        ai_response: str,
        metadata: Dict[str, Any]
    ):
        """
        保存交互记录（双写策略）
        
        Args:
            user_id: 用户ID
            session_id: 会话ID
            user_message: 用户消息
            ai_response: AI响应
            metadata: 元数据（模型、工具、评分等）
        """
        # 1. 保存到MySQL（结构化）
        await self._save_to_mysql(
            user_id, session_id, user_message, ai_response, metadata
        )
        
        # 2. 保存到Qdrant（向量化）
        await self._save_to_qdrant(
            user_id, user_message, ai_response, metadata
        )
    
    async def _save_to_mysql(
        self,
        user_id: str,
        session_id: str,
        user_message: str,
        ai_response: str,
        metadata: Dict[str, Any]
    ):
        """保存到MySQL（结构化存储）"""
        await self.backend_client.save_chat_session({
            "user_id": user_id,
            "session_id": session_id,
            "user_message": user_message,
            "ai_response": ai_response,
            "model_used": metadata.get("model_used"),
            "tools_called": metadata.get("tools_called", []),
            "quality_score": metadata.get("quality_score"),
            "user_rating": metadata.get("user_rating"),
            "created_at": datetime.now().isoformat()
        })
    
    async def _save_to_qdrant(
        self,
        user_id: str,
        user_message: str,
        ai_response: str,
        metadata: Dict[str, Any]
    ):
        """保存到Qdrant（向量化存储）"""
        # 向量化用户消息
        query_vector = self.embedding_model.encode(user_message)
        
        await self.qdrant_helper.add_conversation({
            "user_query": user_message,
            "ai_response": ai_response,
            "vector": query_vector.tolist(),
            "metadata": {
                "user_id": user_id,
                "user_rating": metadata.get("user_rating"),
                "quality_score": metadata.get("quality_score"),
                "model_used": metadata.get("model_used"),
                "timestamp": datetime.now().isoformat()
            }
        })
```

CREATE TABLE chat_sessions (
    id BIGINT PRIMARY KEY AUTO_INCREMENT,
    user_id VARCHAR(255) NOT NULL,
    session_id VARCHAR(255) NOT NULL,

);

### BackendClient实现

```python
# src/applications/fitness/clients/backend_client.py

class BackendClient:
    async def save_chat_session(self, session_data: Dict[str, Any]) -> Dict[str, Any]:
        """
        保存会话记录到MySQL
        
        Args:
            session_data: 会话数据
            
        Returns:
            保存结果
        """
        url = f"{self.config.base_url}/api/internal/chat-sessions"
        
        try:
            response = await self.client.post(url, json=session_data)
            response.raise_for_status()
            
            result = response.json()
            logger.info(f"会话记录已保存: session_id={session_data['session_id']}")
            
            return result
            
        except httpx.HTTPError as e:
            logger.error(f"保存会话记录失败: {e}")
            raise BackendAPIError(f"Failed to save chat session: {e}")
```

### 查询接口

class BackendClient:
    async def get_chat_history(
        self,
        user_id: str,
        limit: int = 10,
        min_rating: Optional[int] = None
    ) -> List[Dict[str, Any]]:
        """
        获取用户的聊天历史
        
        Args:
            user_id: 用户ID
            limit: 返回数量限制
            min_rating: 最低评分过滤
            
        Returns:
            聊天历史列表
        """
        url = f"{self.config.base_url}/api/internal/chat-sessions/{user_id}"
        params = {"limit": limit}
        
        if min_rating is not None:
            params["min_rating"] = min_rating
        
        try:
            response = await self.client.get(url, params=params)
            response.raise_for_status()
            
            return response.json()
            
        except httpx.HTTPError as e:
            logger.error(f"获取聊天历史失败: {e}")
            return []

## Qdrant存储实现

### Collection配置

```python
# src/framework/retrieval/qdrant_helper.py

class QdrantHelper:
    def __init__(self):
        self.client = QdrantClient(
            host=os.getenv("QDRANT_HOST", "localhost"),
            port=int(os.getenv("QDRANT_PORT", 6333))
        )
        self.collection_name = "chat_conversations"
        self.vector_size = 1024  # BGE-M3向量维度
        
        # 初始化collection
        self._init_collection()
    
    def _init_collection(self):
        """初始化Qdrant collection"""
        try:
            self.client.get_collection(self.collection_name)
            logger.info(f"Collection已存在: {self.collection_name}")
        except Exception:
            # Collection不存在，创建新的
            self.client.create_collection(
                collection_name=self.collection_name,
                vectors_config=VectorParams(
                    size=self.vector_size,
                    distance=Distance.COSINE
                )
            )
            logger.info(f"创建新Collection: {self.collection_name}")
```

### 添加对话记录

```python
class QdrantHelper:
    async def add_conversation(self, conversation_data: Dict[str, Any]) -> str:
        """
        添加对话记录到Qdrant
        
        Args:
            conversation_data: 对话数据
            {
                "user_query": str,
                "ai_response": str,
                "vector": List[float],
                "metadata": Dict
            }
            
        Returns:
            point_id: 插入的点ID
        """
        point_id = str(uuid.uuid4())
        
        point = PointStruct(
            id=point_id,
            vector=conversation_data["vector"],
            payload={
                "user_query": conversation_data["user_query"],
                "ai_response": conversation_data["ai_response"],
                "user_id": conversation_data["metadata"].get("user_id"),
                "user_rating": conversation_data["metadata"].get("user_rating"),
                "quality_score": conversation_data["metadata"].get("quality_score"),
                "model_used": conversation_data["metadata"].get("model_used"),
                "timestamp": conversation_data["metadata"].get("timestamp")
            }
        )
        
        self.client.upsert(
            collection_name=self.collection_name,
            points=[point]
        )
        
        logger.info(f"对话记录已添加到Qdrant: {point_id}")
        return point_id
```

### Few-Shot检索

```python
class QdrantHelper:
    async def search_similar_conversations(
        self,
        query_vector: List[float],
        top_k: int = 5,
        min_similarity: float = 0.6,
        min_rating: Optional[int] = None
    ) -> List[Dict[str, Any]]:
        """
        检索相似对话（用于Few-Shot）
        
        Args:
            query_vector: 查询向量
            top_k: 返回数量
            min_similarity: 最低相似度阈值
            min_rating: 最低评分过滤
            
        Returns:
            相似对话列表
        """
        # 构建过滤条件
        filter_conditions = []
        
        if min_rating is not None:
            filter_conditions.append(
                FieldCondition(
                    key="user_rating",
                    match=MatchValue(value=min_rating),
                    range=Range(gte=min_rating)
                )
            )
        
        # 执行搜索
        search_result = self.client.search(
            collection_name=self.collection_name,
            query_vector=query_vector,
            limit=top_k,
            query_filter=Filter(must=filter_conditions) if filter_conditions else None,
            score_threshold=min_similarity
        )
        
        # 格式化结果
        conversations = []
        for hit in search_result:
            conversations.append({
                "user_query": hit.payload["user_query"],
                "ai_response": hit.payload["ai_response"],
                "similarity": hit.score,
                "user_rating": hit.payload.get("user_rating"),
                "quality_score": hit.payload.get("quality_score")
            })
        
        logger.info(f"检索到{len(conversations)}条相似对话")
        return conversations
```

## 双写策略的优势

### 1. MySQL优势（结构化查询）

```python
# 查询用户的所有高分对话
SELECT * FROM chat_sessions
WHERE user_id = 'user_123'
  AND user_rating >= 4
ORDER BY created_at DESC
LIMIT 10;

# 统计模型使用情况
SELECT model_used, COUNT(*) as count
FROM chat_sessions
WHERE created_at >= DATE_SUB(NOW(), INTERVAL 7 DAY)
GROUP BY model_used;

# 分析工具调用频率
SELECT 
    JSON_EXTRACT(tools_called, '$[*]') as tool,
    COUNT(*) as frequency
FROM chat_sessions
GROUP BY tool
ORDER BY frequency DESC;
```

### 3. 互补性

| 特性 | MySQL | Qdrant |
|------|-------|--------|
| **查询方式** | 结构化SQL | 向量相似度 |
| **适用场景** | 精确查询、统计分析 | 语义检索、Few-Shot |
| **查询速度** | 快（索引） | 快（向量索引） |
| **存储成本** | 低 | 中 |
| **扩展性** | 垂直扩展 | 水平扩展 |

## 数据一致性保证

### 1. 事务性保证

```python
async def save_interaction(self, ...):
    """保存交互记录（带事务保证）"""
    mysql_success = False
    qdrant_success = False
    
    try:
        # 1. 保存到MySQL
        await self._save_to_mysql(...)
        mysql_success = True
        
        # 2. 保存到Qdrant
        await self._save_to_qdrant(...)
        qdrant_success = True
        
    except Exception as e:
        logger.error(f"保存交互记录失败: {e}")
        
        # 回滚策略
        if mysql_success and not qdrant_success:
            logger.warning("Qdrant保存失败，但MySQL已保存")
            # 可以稍后重试Qdrant
            await self._schedule_qdrant_retry(...)
        
        raise
```

### 2. 重试机制

```python
class QdrantHelper:
    async def add_conversation_with_retry(
        self,
        conversation_data: Dict[str, Any],
        max_retries: int = 3
    ) -> str:
        """带重试的对话记录添加"""
        for attempt in range(max_retries):
            try:
                return await self.add_conversation(conversation_data)
            except Exception as e:
                if attempt < max_retries - 1:
                    logger.warning(f"Qdrant保存失败，重试 ({attempt + 1}/{max_retries})")
                    await asyncio.sleep(1 * (attempt + 1))
                else:
                    logger.error(f"Qdrant保存失败，已达最大重试次数: {e}")
                    raise
```

## 性能优化

### 1. 批量写入

```python
class QdrantHelper:
    async def batch_add_conversations(
        self,
        conversations: List[Dict[str, Any]],
        batch_size: int = 100
    ):
        """批量添加对话记录"""
        points = []
        
        for conv in conversations:
            point_id = str(uuid.uuid4())
            points.append(PointStruct(
                id=point_id,
                vector=conv["vector"],
                payload={...}
            ))
        
        # 分批插入
        for i in range(0, len(points), batch_size):
            batch = points[i:i + batch_size]
            self.client.upsert(
                collection_name=self.collection_name,
                points=batch
            )
            logger.info(f"批量插入{len(batch)}条记录")
```

### 2. 异步写入

```python
async def save_interaction_async(self, ...):
    """异步保存交互记录（不阻塞主流程）"""
    # 创建后台任务
    asyncio.create_task(self._save_to_mysql(...))
    asyncio.create_task(self._save_to_qdrant(...))
    
    # 立即返回，不等待保存完成
    logger.info("交互记录已提交后台保存")
```
