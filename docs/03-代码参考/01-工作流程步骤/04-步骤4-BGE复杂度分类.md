### 复杂度分类器

```python
class QueryComplexityClassifier:
    """
    查询复杂度分类器
    
    使用BGE-M3模型判断查询是否复杂
    """
    
    def __init__(self, embedding_model, qdrant_client):
        self.embedding_model = embedding_model
        self.qdrant_client = qdrant_client
        self.collection_name = "query_complexity_examples"
    
    def classify_complexity(
        self,
        query: str
    ) -> Tuple[bool, float, str]:
        """
        分类查询复杂度
        
        Args:
            query: 用户查询
            
        Returns:
            (is_complex, similarity, reason)
            - is_complex: 是否复杂
            - similarity: 与历史案例的相似度 (0-1)
            - reason: 分类原因
        """
        # 1. 向量化查询
        query_vector = self.embedding_model.encode(query)
        
        # 2. 检索历史复杂查询案例
        complex_examples = self._search_complex_examples(query_vector)
        
        # 3. 计算相似度
        if complex_examples:
            max_similarity = max(ex["similarity"] for ex in complex_examples)
        else:
            max_similarity = 0.0
        
        # 4. 判断复杂度
        is_complex = max_similarity >= 0.7
        
        # 5. 生成原因
        if is_complex:
            reason = f"与复杂查询案例相似度{max_similarity:.2f}，需要专业规划"
        else:
            reason = f"相似度{max_similarity:.2f}，可使用学生模型"
        
        logger.info(f"复杂度分类: {is_complex}, 相似度: {max_similarity:.2f}")
        
        return is_complex, max_similarity, reason
    
    def _search_complex_examples(
        self,
        query_vector: np.ndarray,
        top_k: int = 5
    ) -> List[Dict[str, Any]]:
        """检索历史复杂查询案例"""
        results = self.qdrant_client.search(
            collection_name=self.collection_name,
            query_vector=query_vector.tolist(),
            limit=top_k,
            query_filter=Filter(
                must=[
                    FieldCondition(
                        key="is_complex",
                        match=MatchValue(value=True)
                    )
                ]
            )
        )
        
        return [
            {
                "query": hit.payload["query"],
                "similarity": hit.score
            }
            for hit in results
        ]
```

## 分类逻辑

### 相似度阈值

```python
# 相似度阈值定义
SIMILARITY_THRESHOLDS = {
    "high": 0.7,      # >= 0.7: 高度相似，复杂查询
    "medium": 0.5,    # 0.5-0.7: 中等相似，灰色地带
    "low": 0.5        # < 0.5: 低相似度，简单查询
}

def classify_by_similarity(similarity: float) -> str:
    """根据相似度分类"""
    if similarity >= SIMILARITY_THRESHOLDS["high"]:
        return "complex"
    elif similarity >= SIMILARITY_THRESHOLDS["medium"]:
        return "uncertain"  # 需要Few-Shot辅助判断
    else:
        return "simple"
```

### 三段式决策（配合步骤5）

```python
def should_use_teacher_model(
    similarity: float,
    few_shot_count: int
) -> bool:
    """
    决定是否使用教师模型
    
    三段式决策：
    1. similarity < 0.5  → 学生模型
    2. similarity >= 0.7 → 教师模型
    3. 0.5 <= similarity < 0.7 → 检查Few-Shot
       - Few-Shot >= 3 → 学生模型
       - Few-Shot < 3  → 教师模型
    """
    if similarity < 0.5:
        return False  # 学生模型
    elif similarity >= 0.7:
        return True   # 教师模型
    else:
        # 灰色地带：检查Few-Shot
        return few_shot_count < 3
```
