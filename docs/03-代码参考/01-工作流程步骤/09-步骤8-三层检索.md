## 概述

步骤8实现三层检索引擎（Vector → Graph → Constraint），确保100%基于真实数据，无幻觉。

### 三层检索架构

```
Layer 1: 向量语义搜索（BGE-M3 + Qdrant）
  ↓ 召回Top 20候选
Layer 2: 图关系查询（Neo4j Cypher）
  ↓ 精确过滤（禁忌、关系验证）
Layer 3: 业务规则（Python逻辑）
  ↓ 最终结果（100%安全可用）
```

### Layer 1: 向量语义搜索

```python
async def _semantic_search(
    self,
    query: str,
    top_k: int = 20
) -> List[Dict[str, Any]]:
    """
    Layer 1: 向量语义搜索
    
    使用BGE-M3模型进行语义检索
    """
    # 向量化查询
    query_vector = self.embedding_model.encode(query)
    
    # Qdrant搜索
    results = self.qdrant_client.search(
        collection_name="fitness_kg",
        query_vector=query_vector.tolist(),
        limit=top_k,
        score_threshold=0.6  # 最低相似度阈值
    )
    
    # 返回候选
    candidates = []
    for hit in results:
        candidates.append({
            "id": hit.payload["id"],
            "name": hit.payload["name"],
            "type": hit.payload["type"],  # Exercise/Muscle/Food
            "similarity": hit.score
        })
    
    logger.info(f"Layer 1召回{len(candidates)}个候选")
    return candidates
```

### Layer 2: 图关系查询

```python
async def _graph_query(
    self,
    candidates: List[Dict[str, Any]],
    user_profile: Dict[str, Any]
) -> List[Dict[str, Any]]:
    """
    Layer 2: 图关系查询
    
    使用Neo4j Cypher精确过滤
    """
    candidate_ids = [c["id"] for c in candidates]
    user_diseases = user_profile.get("health_conditions", [])
    
    # Cypher查询：过滤禁忌动作
    query = """
    MATCH (e:Exercise)
    WHERE e.id IN $candidate_ids
    
    // 检查禁忌关系
    OPTIONAL MATCH (e)-[:CONTRAINDICATED_FOR]->(d:Disease)
    WHERE d.name IN $user_diseases
    
    // 只返回无禁忌的动作
    WITH e, collect(d.name) as contraindications
    WHERE size(contraindications) = 0
    
    // 获取目标肌肉
    OPTIONAL MATCH (e)-[:TARGETS_PRIMARY]->(m:Muscle)
    
    RETURN e.id as id, e.name as name, collect(m.name) as target_muscles
    """
    
    results = self.neo4j_client.execute_query(query, {
        "candidate_ids": candidate_ids,
        "user_diseases": user_diseases
    })
    
    logger.info(f"Layer 2过滤后剩余{len(results)}个动作")
    return results
```

### Layer 3: 业务规则

```python
async def _apply_business_rules(
    self,
    candidates: List[Dict[str, Any]],
    user_profile: Dict[str, Any]
) -> List[Dict[str, Any]]:
    """
    Layer 3: 业务规则
    
    应用专业逻辑过滤
    """
    filtered = []
    
    for exercise in candidates:
        # 规则1: 训练水平过滤
        if user_profile["training_level"] == "beginner":
            if exercise.get("difficulty") == "advanced":
                continue
        
        # 规则2: 器械可用性过滤
        required_equipment = exercise.get("equipment")
        available_equipment = user_profile.get("available_equipment", [])
        if required_equipment not in available_equipment:
            continue
        
        # 规则3: 恢复能力调整
        if user_profile.get("recovery_capacity", 1.0) < 0.6:
            # 降低训练容量
            exercise["recommended_sets"] = int(exercise["recommended_sets"] * 0.7)
        
        filtered.append(exercise)
    
    logger.info(f"Layer 3最终返回{len(filtered)}个动作")
    return filtered
```
