### 权限检查实现

```python
class BackendClient:
    async def get_user_membership(self, user_id: str) -> MembershipPermissions:
        """
        获取用户会员权限
        
        Args:
            user_id: 用户ID
            
        Returns:
            MembershipPermissions: 会员权限对象
        """
        url = f"{self.config.base_url}/api/internal/membership/{user_id}"
        
        try:
            response = await self.client.get(url)
            response.raise_for_status()
            data = response.json()
            
            return MembershipPermissions(
                user_id=data["user_id"],
                tier=MembershipTier(data["tier"]),
                status=data["status"],
                permissions=data["permissions"],
                started_at=data.get("started_at"),
                expired_at=data.get("expired_at")
            )
            
        except httpx.HTTPError as e:
            logger.error(f"获取会员权限失败: {e}")
            # 降级为免费用户
            return self._get_default_free_membership(user_id)
```

### 权限验证

```python
class PermissionChecker:
    def __init__(self, membership: MembershipPermissions):
        self.membership = membership
    
    def can_use_feature(self, feature: str) -> bool:
        """检查是否可以使用某个功能"""
        tier_features = MEMBERSHIP_FEATURES.get(self.membership.tier, {})
        return tier_features.get(feature, False)
    
    def check_daily_limit(self, current_queries: int) -> bool:
        """检查是否超过每日查询限制"""
        tier_features = MEMBERSHIP_FEATURES.get(self.membership.tier, {})
        max_queries = tier_features.get("max_daily_queries", 10)
        
        if max_queries == -1:  # 无限制
            return True
        
        return current_queries < max_queries
    
    def get_upgrade_suggestion(self, requested_feature: str) -> Optional[str]:
        """获取升级建议"""
        if self.can_use_feature(requested_feature):
            return None
        
        # 查找最低可用等级
        for tier in [MembershipTier.WARMHEART, MembershipTier.ENERGY]:
            if MEMBERSHIP_FEATURES[tier].get(requested_feature):
                return f"该功能需要{tier.value}会员，请升级后使用"
        
        return "该功能暂不可用"
```

        # ...
        
        # 步骤3: 会员权限检查
        membership = await self.backend_client.get_user_membership(user_id)
        permission_checker = PermissionChecker(membership)
