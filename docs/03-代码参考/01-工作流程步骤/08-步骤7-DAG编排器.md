### DAG编排器

```python
class EnhancedDAGOrchestrator:
    """
    增强型DAG编排器
    
    职责：
    - 执行DAG模板定义的工具链
    - 自动处理工具依赖关系
    - 并行优化（无依赖工具并行执行）
    - 缓存管理（避免重复调用）
    """
    
    async def execute_dag(
        self,
        template: DAGTemplate,
        context: Dict[str, Any]
    ) -> Dict[str, Any]:
        """
        执行DAG
        
        Args:
            template: DAG模板
            context: 执行上下文（包含用户档案等）
            
        Returns:
            工具执行结果
        """
        execution_id = str(uuid.uuid4())
        logger.info(f"开始执行DAG: {template.template_id}, execution_id={execution_id}")
        
        # 1. 构建DAG任务
        dag_tasks = await self._build_dag_tasks(template, context)
        
        # 2. 拓扑排序（Kahn算法）
        execution_levels = self._topological_sort(dag_tasks, template.tool_dependencies)
        
        # 3. 并行优化
        optimized_levels = self._optimize_parallel_execution(execution_levels)
        
        # 4. 执行DAG
        results = await self._execute_dag_levels(execution_id, optimized_levels, context)
        
        logger.info(f"DAG执行完成: {execution_id}")
        return results
    
    def _topological_sort(
        self,
        tasks: List[DAGTask],
        dependencies: Dict[str, List[str]]
    ) -> List[List[DAGTask]]:
        """
        拓扑排序（Kahn算法）
        
        将DAG任务分层，同一层的任务可以并行执行
        
        Returns:
            [[Level 0 tasks], [Level 1 tasks], [Level 2 tasks], ...]
        """
        # 计算入度
        in_degree = {task.tool_name: 0 for task in tasks}
        for tool_name, deps in dependencies.items():
            in_degree[tool_name] = len(deps)
        
        # 初始化队列（入度为0的任务）
        queue = [task for task in tasks if in_degree[task.tool_name] == 0]
        levels = []
        
        while queue:
            # 当前层的所有任务
            current_level = queue.copy()
            levels.append(current_level)
            queue.clear()
            
            # 处理当前层的任务
            for task in current_level:
                # 减少依赖此任务的其他任务的入度
                for other_task in tasks:
                    if task.tool_name in dependencies.get(other_task.tool_name, []):
                        in_degree[other_task.tool_name] -= 1
                        if in_degree[other_task.tool_name] == 0:
                            queue.append(other_task)
        
        return levels
    
    async def _execute_dag_levels(
        self,
        execution_id: str,
        levels: List[List[DAGTask]],
        context: Dict[str, Any]
    ) -> Dict[str, Any]:
        """
        执行DAG层级
        
        同一层的任务并行执行
        """
        all_results = {}
        
        for level_idx, level_tasks in enumerate(levels):
            logger.info(f"执行Level {level_idx}: {[t.tool_name for t in level_tasks]}")
            
            # 并行执行同一层的所有任务
            level_results = await asyncio.gather(*[
                self._execute_single_task(task, context, all_results)
                for task in level_tasks
            ])
            
            # 合并结果
            for task, result in zip(level_tasks, level_results):
                all_results[task.tool_name] = result
        
        return all_results
    
    async def _execute_single_task(
        self,
        task: DAGTask,
        context: Dict[str, Any],
        previous_results: Dict[str, Any]
    ) -> Any:
        """执行单个任务"""
        # 检查缓存
        cache_key = self._get_cache_key(task, context)
        if cache_key in self.cache:
            logger.info(f"从缓存返回: {task.tool_name}")
            return self.cache[cache_key]
        
        # 准备工具参数
        tool_args = self._prepare_tool_args(task, context, previous_results)
        
        # 调用MCP工具
        result = await self.mcp_tool_manager.call_tool(
            tool_name=task.tool_name,
            arguments=tool_args
        )
        
        # 更新缓存
        self.cache[cache_key] = result
        
        return result
```

## 关键特性

### 1. 自动并行优化

```python
# 无依赖的工具自动并行执行
# Level 1的3个工具同时执行，节省时间

# 串行执行（旧方式）：
total_time = 200ms + 150ms + 100ms = 450ms

# 并行执行（新方式）：
total_time = max(200ms, 150ms, 100ms) = 200ms

# 性能提升：450ms → 200ms（节省55%时间）
```

### 2. 智能缓存

```python
# 相同参数的工具调用自动缓存
cache_key = f"{tool_name}:{hash(tool_args)}"

# 第一次调用：执行工具（200ms）
result1 = await execute_tool("contraindications_checker", args)

# 第二次调用：从缓存返回（1ms）
result2 = await execute_tool("contraindications_checker", args)
```

### 3. 依赖保证

```python
# 程序自动保证依赖关系
# 禁忌检查必须在动作选择之前执行

# ✅ 正确顺序（自动保证）
contraindications = await check_contraindications()
exercises = await select_exercises(contraindications)

# ❌ 错误顺序（程序阻止）
# exercises = await select_exercises()  # 缺少依赖，无法执行
```
