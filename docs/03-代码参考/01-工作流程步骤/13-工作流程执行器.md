## 概述

工作流程执行器是DAML-RAG系统的入口点，负责初始化所有组件并执行完整的11步工作流程。

### 文件位置

- **主文件**: `src/applications/fitness/workflow_executor.py`
- **依赖**: 所有框架层和应用层组件

### 核心类：WorkflowExecutor

```python
class WorkflowExecutor:
    """
    工作流程执行器
    
    职责：
    - 初始化所有框架层和应用层组件
    - 执行完整的11步工作流程
    - 协调各个步骤的执行
    """
    
    def __init__(self):
        # 初始化框架层组件
        self.embedding_model = self._init_embedding_model()
        self.neo4j_client = self._init_neo4j_client()
        self.qdrant_client = self._init_qdrant_client()
        self.llm_client = self._init_llm_client()
        
        # 初始化应用层组件
        self.backend_client = BackendClient()
        self.query_complexity_classifier = QueryComplexityClassifier()
        self.llm_decision_engine = LLMDecisionEngine()
        self.dag_orchestrator = EnhancedDAGOrchestrator()
        self.llm_analysis_engine = LLMAnalysisEngine()
        
        # 初始化MCP工具注册表
        self.mcp_tool_registry = MCPToolRegistry()
        self._register_all_tools()
    
    async def execute(
        self,
        user_id: str,
        message: str,
        session_id: str
    ) -> Dict[str, Any]:
        """
        执行完整工作流程
        
        Args:
            user_id: 用户ID
            message: 用户消息
            session_id: 会话ID
            
        Returns:
            AI响应和元数据
        """
        # 步骤1: 预加载用户档案
        user_profile = await self.backend_client.get_user_profile(user_id)
        
        # 步骤2: 会话记录存储（异步）
        # 步骤3: 会员权限检查
        membership = await self.backend_client.get_user_membership(user_id)
        
        # 步骤4: BGE复杂度分类
        is_complex, similarity, reason = self.query_complexity_classifier.classify_complexity(message)
        
        # 步骤5: 智能模型选择
        # 步骤6: Few-Shot检索
        few_shot_examples = await self._retrieve_few_shot_examples(message)
        
        # 步骤6.5: LLM选择DAG方案
        dag_template = await self.llm_decision_engine.select_dag_template(
            message, user_profile
        )
        
        # 步骤7: DAG编排器执行
        tool_results = await self.dag_orchestrator.execute_dag(
            dag_template, {"user_profile": user_profile}
        )
        
        # 步骤8: 三层检索（在DAG中执行）
        # 步骤9: 工具结果汇总
        aggregated_results = self._aggregate_results(tool_results)
        
        # 步骤10: LLM深度分析
        ai_response = await self.llm_analysis_engine.analyze_and_synthesize(
            message, user_profile, aggregated_results
        )
        
        # 步骤11: 交互记录
        await self._save_interaction(
            user_id, session_id, message, ai_response
        )
        
        return {
            "response": ai_response,
            "metadata": {
                "model_used": "deepseek" if is_complex else "anthropic-claude",
                "similarity": similarity,
                "tools_called": list(tool_results.keys())
            }
        }
```

## 组件初始化

### 框架层组件

```python
def _init_embedding_model(self):
    """初始化GTE-Large-zh嵌入模型"""
    from sentence_transformers import SentenceTransformer
    return SentenceTransformer('BAAI/bge-m3')

def _init_neo4j_client(self):
    """初始化Neo4j客户端"""
    from src.framework.clients.neo4j_client import Neo4jClient
    return Neo4jClient(
        uri=os.getenv("NEO4J_URI"),
        user=os.getenv("NEO4J_USER"),
        password=os.getenv("NEO4J_PASSWORD")
    )

def _init_qdrant_client(self):
    """初始化Qdrant客户端"""
    from qdrant_client import QdrantClient
    return QdrantClient(
        host=os.getenv("QDRANT_HOST"),
        port=int(os.getenv("QDRANT_PORT"))
    )

def _init_llm_client(self):
    """初始化LLM客户端"""
    from src.framework.clients.llm_client import LLMClient
    return LLMClient()
```
