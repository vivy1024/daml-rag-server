```python
    template_id: str,

    user_profile: dict,

    """
    根据DAG模板构建提示词
    
    Args:
        template_id: DAG模板ID（如"greeting", "quick_consultation"）
        response_hint: 模板的响应提示（从template.response_hint获取）

    Returns:
        str: 构建好的system_prompt
    """
```

### 实现逻辑

#### 1. Greeting模板（特殊处理）

```python
if template_id == "greeting":

响应要求: {response_hint}

```

```python
if template_id == "quick_consultation":
```

### 实现逻辑

```python
def get_llm_config_by_template(template_id: str) -> dict:
    """根据DAG模板获取LLM配置"""
    
    # 模板ID到LLM配置的映射
    CONFIG_MAP = {
        # 问候闲聊：极简配置
        "greeting": {
            "max_tokens": 100,
            "temperature": 0.8  # 较高温度，更自然友好
        },
        
        # 快速咨询：简洁配置
        "quick_consultation": {
            "max_tokens": 800,
            "temperature": 0.6  # 中等温度，平衡创造性和准确性
        },
        
        # 动作优化：实用配置
        "exercise_optimization": {
            "max_tokens": 1500,
            "temperature": 0.6
        },
        
        # 安全评估：严谨配置
        "safety_assessment": {
            "max_tokens": 2000,
            "temperature": 0.5  # 较低温度，更严谨准确
        },
        
        # 完整训练计划：详细配置
        "complete_training_plan": {
            "max_tokens": 3000,
            "temperature": 0.7  # 平衡专业性和创造性
        },
        
        # 营养规划：专业配置
        "nutrition_planning": {
            "max_tokens": 2500,
            "temperature": 0.6
        },
        
        # 综合健身方案：全面配置
        "comprehensive_fitness": {
            "max_tokens": 4000,
            "temperature": 0.7
        },
        
        # 进展分析：数据驱动配置
        "progress_analysis": {
            "max_tokens": 2000,
            "temperature": 0.6
        },
        
        # 康复训练：谨慎配置
        "rehabilitation_training": {
            "max_tokens": 2500,
            "temperature": 0.5  # 较低温度，更谨慎
        }
    }
    
    # 返回配置，如果模板ID不存在则使用默认配置
    return CONFIG_MAP.get(template_id, {
        "max_tokens": 2000,
        "temperature": 0.7
    })
```

### 配置说明

#### Max Tokens配置原则

| 模板类型 | max_tokens | 说明 |
|---------|-----------|------|
| **Greeting** | 100 | 1-2句话，极简回应 |
| **Quick Consultation** | 800 | 2-3段话，简洁实用 |
| **Exercise Optimization** | 1500 | 动作推荐+说明 |
| **Safety Assessment** | 2000 | 安全评估+建议 |
| **Complete Training Plan** | 3000 | 完整训练计划 |
| **Nutrition Planning** | 2500 | 营养方案+膳食计划 |
| **Comprehensive Fitness** | 4000 | 训练+营养完整方案 |
| **Progress Analysis** | 2000 | 数据分析+优化建议 |
| **Rehabilitation Training** | 2500 | 康复计划+注意事项 |

#### Temperature配置原则

| Temperature | 适用场景 | 特点 |
|------------|---------|------|
| **0.8** | Greeting | 更自然、更友好、更有变化 |
| **0.7** | 训练计划、综合方案 | 平衡专业性和创造性 |
| **0.6** | 快速咨询、营养规划 | 平衡准确性和灵活性 |
| **0.5** | 安全评估、康复训练 | 更严谨、更准确、更保守 |

```python
    # 验证响应内容
    assert "训练计划" not in response
    assert "动作" not in response
```
