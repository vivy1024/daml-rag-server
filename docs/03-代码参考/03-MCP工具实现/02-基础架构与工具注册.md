## BaseMCPTool基类

所有MCP工具必须继承此基类并实现抽象方法。

**代码路径**: `mcp_tools/base_tool.py`

### 核心接口

```python
from abc import ABC, abstractmethod
from typing import Dict, Any, Optional, List
from pydantic import BaseModel

class BaseMCPTool(ABC):
    """MCP工具基类"""
    
    def __init__(
        self,
        neo4j_client,
        qdrant_client,
        three_layer_engine,
        logger: Optional[logging.Logger] = None
    ):
        self.neo4j_client = neo4j_client
        self.qdrant_client = qdrant_client
        self.three_layer_engine = three_layer_engine
        self.logger = logger or logging.getLogger(self.__class__.__name__)
    
    @abstractmethod
    def get_name(self) -> str:
        """返回工具名称"""
        pass
    
    @abstractmethod
    def get_description(self) -> str:
        """返回工具描述"""
        pass
    
    @abstractmethod
    def get_category(self) -> str:
        """返回工具分类（exercise/training/safety/nutrition/learning）"""
        pass
    
    @abstractmethod
    def get_input_schema(self) -> type[BaseModel]:
        """返回输入Schema（Pydantic模型类）"""
        pass
    
    @abstractmethod
    def get_output_schema(self) -> type[BaseModel]:
        """返回输出Schema（Pydantic模型类）"""
        pass
    
    @abstractmethod
    async def execute(self, input_data: Dict[str, Any]) -> Dict[str, Any]:
        """执行工具逻辑"""
        pass
```

### 性能监控包装器

```python
async def execute_with_monitoring(
    self,
    input_data: Dict[str, Any]
) -> Dict[str, Any]:
    """带性能监控的执行包装器"""
    start_time = time.time()
    tool_name = self.get_name()
    
    try:
        # 验证输入
        input_schema = self.get_input_schema()
        validated_input = input_schema(**input_data)
        
        # 执行工具
        result = await self.execute(validated_input.model_dump())
        
        # 添加元数据
        execution_time_ms = (time.time() - start_time) * 1000
        result.setdefault("metadata", {})
        result["metadata"]["execution_time_ms"] = execution_time_ms
        result["metadata"]["timestamp"] = datetime.now().isoformat()
        result["metadata"]["tool_name"] = tool_name
        
        return result
        
    except Exception as e:
        # 错误处理
        return {
            "success": False,
            "tool_name": tool_name,
            "error": {
                "code": "TOOL_EXECUTION_ERROR",
                "message": str(e),
                "type": type(e).__name__
            }
        }
```

## 异常类型

**代码路径**: `mcp_tools/exceptions.py`

### 异常层次结构

```python
class ToolError(Exception):
    """工具基础异常"""
    def __init__(self, message: str, tool_name: str = None, context: dict = None):
        self.message = message
        self.tool_name = tool_name
        self.context = context or {}

class ToolValidationError(ToolError):
    """输入验证错误"""
    pass

class ToolTimeoutError(ToolError):
    """执行超时错误"""
    def __init__(self, message: str, tool_name: str = None, timeout_ms: float = None):
        super().__init__(message, tool_name)
        self.timeout_ms = timeout_ms

class ToolConnectionError(ToolError):
    """数据库连接错误"""
    def __init__(self, message: str, tool_name: str = None, service: str = None):
        super().__init__(message, tool_name)
        self.service = service

class ToolExecutionError(ToolError):
    """执行错误"""
    def __init__(self, message: str, tool_name: str = None, stage: str = None):
        super().__init__(message, tool_name)
        self.stage = stage
```

## MCPToolRegistry注册表

**代码路径**: `mcp_tools/registry.py`

### 核心功能

```python
class MCPToolRegistry:
    """MCP工具注册表"""
    
    def __init__(self):
        self.tools: Dict[str, BaseMCPTool] = {}
        self.metadata: Dict[str, ToolMetadata] = {}
        self.performance_stats: Dict[str, Dict[str, Any]] = {}
    
    def register_tool(self, tool: BaseMCPTool) -> None:
        """注册工具"""
        tool_name = tool.get_name()
        
        if tool_name in self.tools:
            raise ValueError(f"工具 {tool_name} 已注册")
        
        self.tools[tool_name] = tool
        self.metadata[tool_name] = tool.get_metadata()
        self.performance_stats[tool_name] = {
            "total_calls": 0,
            "success_count": 0,
            "error_count": 0,
            "total_duration_ms": 0.0,
            "avg_duration_ms": 0.0
        }
    
    async def call_tool(
        self,
        tool_name: str,
        input_data: Dict[str, Any]
    ) -> Dict[str, Any]:
        """调用工具"""
        tool = self.get_tool(tool_name)
        if not tool:
            raise ValueError(f"工具不存在: {tool_name}")
        
        # 更新统计
        self.performance_stats[tool_name]["total_calls"] += 1
        
        # 执行工具
        result = await tool.execute_with_monitoring(input_data)
        
        # 更新性能统计
        if result.get("success"):
            self.performance_stats[tool_name]["success_count"] += 1
        else:
            self.performance_stats[tool_name]["error_count"] += 1
        
        return result
```
